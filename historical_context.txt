    pseudo_rp:  pseudoRp.trim(),
    temoignage: temoignage?.trim() || null,
  });
  if (error) {
    if (error.message.includes('unique') || error.code === '23505')
      throw new Error('Vous avez déjà rejoint ce dossier.');
    throw new Error(error.message);
  }
}

/**
 * Quitte un dossier collectif.
 */
export async function leavePlainte(plainteId: string): Promise<void> {
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) throw new Error('Non authentifié.');
  const { error } = await supabase
    .from('co_plaignants')
    .delete()
    .eq('plainte_id', plainteId)
    .eq('user_id', user.id);
  if (error) throw new Error(error.message);
}

   381   afterId?:    string | null;
   382   categoryId?: string;
   383   status?:     string;
   384   sortBy?:     'date' | 'votes';
   385   search?:     string;

   388 
   389   let q = supabase
   390     .from('plaintes')
   391     .select(`
   392       *,
   393       profiles!plaintes_user_id_fkey(username),

   396       commentaires(id)

   399     .limit(lim + 1);
   400 
   401   if (categoryId)                  q = q.eq('category_id', categoryId);
   402   if (status && status !== 'all')  q = q.eq('status', status);
   403   if (afterId) {

   405     const { data: cur } = await supabase

   407     if (cur) q = q.lt('created_at', cur.created_at);
   408   }

   411   const rows = Array.isArray(data) ? data : [];
   412   const hasMore = rows.length > lim;
   413   const slice   = hasMore ? rows.slice(0, lim) : rows;
   414   let plaintes  = normalizePlaintes(slice);
   415 
   416   if (search) {
   417     const s = search.toLowerCase();
   418     plaintes = plaintes.filter((p) =>
   419       p.game_server_name.toLowerCase().includes(s) ||
   420       p.admin_name.toLowerCase().includes(s) ||
   421       p.description.toLowerCase().includes(s)
   422     );
   423   }
   424   if (sortBy === 'votes') {
   425     plaintes.sort((a, b) =>
   426       ((b.upvotes ?? 0) - (b.downvotes ?? 0)) - ((a.upvotes ?? 0) - (a.downvotes ?? 0))
   427     );
   428   }
   429

   441   const votes   = Array.isArray(row.votes) ? row.votes : [];
   442   const comments = Array.isArray(row.commentaires) ? row.commentaires : [];

// ── Cache en mémoire partagé entre tous les modules API ──────────────────────

const store: Record<string, { data: unknown; expireAt: number }> = {};

export function getCached<T>(key: string): T | null {
  const entry = store[key];
  if (entry && Date.now() < entry.expireAt) return entry.data as T;
  return null;
}

export function setCache(key: string, data: unknown, ttlSeconds: number): void {
  store[key] = { data, expireAt: Date.now() + ttlSeconds * 1000 };
}

export function invalidateCache(prefix: string): void {
  for (const key of Object.keys(store)) {
    if (key.startsWith(prefix)) delete store[key];
  }
}

/** Convertit un nom de fichier en snake_case sûr pour Storage. */
export function snakeCase(name: string): string {
  return name.trim().toLowerCase().replace(/[^a-z0-9]+/g, '_');
}

/** Hache une chaîne en SHA-256 (hex). */
export async function sha256(text: string): Promise<string> {
  const buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(text));
  return Array.from(new Uint8Array(buf))
    .map((b) => b.toString(16).padStart(2, '0'))
    .join('');
}

/** Valide qu'une chaîne est une URL http(s) valide. */
export function isValidUrl(url: string): boolean {
  try {
    const u = new URL(url.trim());
    return u.protocol === 'http:' || u.protocol === 'https:';
  } catch {
    return false;
  }
}

// eslint-disable-next-line @typescript-eslint/no-explicit-any
export function normalizePlainte(row: any): Plainte {
  const votes    = Array.isArray(row.votes) ? row.votes : [];
  const comments = Array.isArray(row.commentaires) ? row.commentaires : [];
  const upvotes   = votes.filter((v: { vote_type: string }) => v.vote_type === 'pour').length;
  const downvotes = votes.filter((v: { vote_type: string }) => v.vote_type === 'contre').length;
  return {
    ...row,
    profiles:      row.profiles   ?? null,
    categories:    row.categories ?? null,
    upvotes,
    downvotes,
    vote_count:    upvotes - downvotes,
    comment_count: comments.length,
  };
}

export function normalizePlaintes(rows: any[]): Plainte[] {
